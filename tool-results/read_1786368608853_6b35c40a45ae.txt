    24→// --- Utility functions ---
    25→function formatSilver(price: number | null): string {
    26→  if (price === null || price === undefined) return '-'
    27→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    28→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    29→  return price.toString()
    30→}
    31→
    32→function formatGoldPrice(price: number): string {
    33→  return price.toLocaleString('fr-FR')
    34→}
    35→
    36→function timeAgo(timestamp: number): string {
    37→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
    38→  if (seconds < 10) return "à l'instant"
    39→  if (seconds < 60) return `il y a ${seconds}s`
    40→  const minutes = Math.floor(seconds / 60)
    41→  if (minutes < 60) return `il y a ${minutes}min`
    42→  const hours = Math.floor(minutes / 60)
    43→  return `il y a ${hours}h`
    44→}

    46→function getTierColor(name: string): string {
    47→  if (name.includes('T8')) return 'text-amber-400'
    48→  if (name.includes('T7')) return 'text-purple-400'
    49→  if (name.includes('T6')) return 'text-orange-400'
    50→  if (name.includes('T5')) return 'text-sky-400'
    51→  return 'text-foreground'
    52→}

    54→function getTierBadgeVariant(name: string): 'default' | 'secondary' | 'outline' | 'destructive' {
    55→  if (name.includes('T8')) return 'default'
    56→  if (name.includes('T7')) return 'secondary'
    57→  return 'outline'
    58→}

    65→function getItemCategory(name: string): string {
    66→  const lower = name.toLowerCase()
    67→  if (lower.includes('minerai') || lower.includes('bois') || lower.includes('fibre') || lower.includes('peau') || lower.includes('pierre')) return 'Ressource'
    68→  if (lower.includes('lingot') || lower.includes('planche') || lower.includes('tissu') || lower.includes('cuir') || lower.includes('bloc de pierre')) return 'Matériau'
    69→  if (lower.includes('armure') || lower.includes('coiffe') || lower.includes('chaussures')) return 'Armure'
    70→  if (lower.includes('sac')) return 'Équipement'
    71→  if (lower.includes('cape')) return 'Équipement'
    72→  if (lower.includes('rune') || lower.includes('âme') || lower.includes('relique')) return 'Rune'
    73→  if (lower.includes('potion')) return 'Potion'
    74→  return 'Autre'
    75→}

    77→function getRenderId(dataProjectId: string): string {
    78→  // Most Data Project IDs match render IDs directly.
    79→  // Only a few items need remapping.
    80→  const m = dataProjectId.match(/^(T\d+_|)(.+)$/)
    81→  if (!m) return dataProjectId
    82→  const tier = m[1] || ''
    83→  const base = m[2]
    84→  const MAP: Record<string, string> = {
    85→    CLOTH: 'CLOTHITEM',
    86→    LEATHER: 'LEATHERITEM',
    87→  }
    88→  return tier + (MAP[base] || base)
    89→}

    91→function ItemIcon({ itemId, size = 32 }: { itemId: string; size?: number }) {
    92→  const [error, setError] = useState(false)
    93→  const renderId = getRenderId(itemId)
    94→  const src = `https://render.albiononline.com/v1/item/${renderId}?quality=1`
    95→  const tier = itemId.match(/T(\d)/)
    96→  const tierNum = tier ? parseInt(tier[1]) : 4
    97→  const colors = ['','oklch(0.7 0.15 250)','oklch(0.65 0.2 200)','oklch(0.7 0.18 40)','oklch(0.7 0.18 40)','oklch(0.75 0.2 80)','oklch(0.7 0.2 300)','oklch(0.65 0.22 300)','oklch(0.8 0.18 80)']
    98→  const color = colors[tierNum] || colors[4]
    99→  if (error) {
   100→    return (
   101→      <div
   102→        className="inline-flex items-center justify-center rounded-md border border-border/30 flex-shrink-0"
   103→        style={{ width: size, height: size, minWidth: size, minHeight: size, backgroundColor: color + '33' }}
   104→        title={renderId}
   105→      >
   106→        <span style={{ fontSize: size * 0.45, color, fontWeight: 700, lineHeight: 1 }}>{tier ? tier[1] : '?'}</span>
   107→      </div>
   108→    )
   109→  }
   110→  return (
   111→    <img
   112→      src={src}
   113→      alt=""
   114→      width={size}
   115→      height={size}
   116→      loading="lazy"
   117→      onError={() => setError(true)}
   118→      className="inline-block rounded-md bg-black/30 border border-border/30 flex-shrink-0"
   119→      style={{ imageRendering: 'auto', minWidth: size, minHeight: size }}
   120→    />
   121→  )
   122→}
