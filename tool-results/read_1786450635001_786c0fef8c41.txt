    21→    21→function formatSilver(price: number | null): string {
    22→    22→  if (price === null || price === undefined) return '-'
    23→    23→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    24→    24→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    25→    25→  return price.toString()
    26→    26→}

    28→    28→function formatGoldPrice(price: number): string {
    29→    29→  return price.toLocaleString('fr-FR')
    30→    30→}

    32→    32→function timeAgo(timestamp: number): string {
    33→    33→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
    34→    34→  if (seconds < 10) return "à l'instant"
    35→    35→  if (seconds < 60) return `il y a ${seconds}s`
    36→    36→  const minutes = Math.floor(seconds / 60)
    37→    37→  if (minutes < 60) return `il y a ${minutes}min`
    38→    38→  const hours = Math.floor(minutes / 60)
    39→    39→  return `il y a ${hours}h`
    40→    40→}

    61→    61→function getItemCategory(name: string): string {
    62→    62→  const lower = name.toLowerCase()
    63→    63→  if (lower.includes('minerai') || lower.includes('bois') || lower.includes('fibre') || lower.includes('peau') || lower.includes('pierre')) return 'Ressource'
    64→    64→  if (lower.includes('lingot') || lower.includes('planche') || lower.includes('tissu') || lower.includes('cuir') || lower.includes('bloc de pierre')) return 'Matériau'
    65→    65→  if (lower.includes('armure') || lower.includes('coiffe') || lower.includes('chaussures')) return 'Armure'
    66→    66→  if (lower.includes('sac')) return 'Équipement'
    67→    67→  if (lower.includes('cape')) return 'Équipement'
    68→    68→  if (lower.includes('rune') || lower.includes('âme') || lower.includes('relique')) return 'Rune'
    69→    69→  if (lower.includes('potion')) return 'Potion'
    70→    70→  return 'Autre'
    71→    71→}

    73→    73→function getRenderId(dataProjectId: string): string {
    74→    74→  // Most Data Project IDs match render IDs directly.
    75→    75→  // Only a few items need remapping.
    76→    76→  const m = dataProjectId.match(/^(T\d+_|)(.+)$/)
    77→    77→  if (!m) return dataProjectId
    78→    78→  const tier = m[1] || ''
    79→    79→  const base = m[2]
    80→    80→  const MAP: Record<string, string> = {
    81→    81→    CLOTH: 'CLOTHITEM',
    82→    82→    LEATHER: 'LEATHERITEM',
    83→    83→  }
    84→    84→  return tier + (MAP[base] || base)
    85→    85→}
