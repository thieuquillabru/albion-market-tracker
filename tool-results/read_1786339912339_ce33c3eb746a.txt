    94→function formatSilver(price: number | null): string {
    95→  if (price === null || price === undefined) return '-'
    96→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    97→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    98→  return price.toString()
    99→}
   100→
   101→function formatGoldPrice(price: number): string {
   102→  return price.toLocaleString('fr-FR')
   103→}
   104→
   105→function timeAgo(timestamp: number): string {
   106→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
   107→  if (seconds < 10) return "\u00e0 l'instant"
   108→  if (seconds < 60) return `il y a ${seconds}s`
   109→  const minutes = Math.floor(seconds / 60)
   110→  if (minutes < 60) return `il y a ${minutes}min`
   111→  const hours = Math.floor(minutes / 60)
   112→  return `il y a ${hours}h`
   113→}

   115→function getTierColor(name: string): string {
   116→  if (name.includes('T8')) return 'text-amber-400'
   117→  if (name.includes('T7')) return 'text-purple-400'
   118→  if (name.includes('T6')) return 'text-orange-400'
   119→  if (name.includes('T5')) return 'text-sky-400'
   120→  return 'text-foreground'
   121→}

   123→function getTierBadgeVariant(name: string): 'default' | 'secondary' | 'outline' | 'destructive' {
   124→  if (name.includes('T8')) return 'default'
   125→  if (name.includes('T7')) return 'secondary'
   126→  return 'outline'
   127→}

   134→function getItemCategory(name: string): string {
   135→  const lower = name.toLowerCase()
   136→  if (lower.includes('sword') || lower.includes('broadsword') || lower.includes('claymore') || lower.includes('dual') || lower.includes('escimitar')) return 'Arme Blanche'
   137→  if (lower.includes('dagger') || lower.includes('katar') || lower.includes('shadow dagger')) return 'Arme Blanche'
   138→  if (lower.includes('staff') || lower.includes('arcane')) return 'B\u00e2ton Magique'
   139→  if (lower.includes('bow') || lower.includes('crossbow') || lower.includes('longbow') || lower.includes('warbow')) return 'Arme \u00e0 Distance'
   140→  if (lower.includes('hammer') || lower.includes('mace') || lower.includes('battleaxe') || lower.includes('halberd') || lower.includes('quarterstaff')) return 'Arme de M\u00eal\u00e9e'
   141→  if (lower.includes('head') || lower.includes('armor') || lower.includes('bag') || lower.includes('cape')) return 'Armure'
   142→  if (lower.includes('ore') || lower.includes('wood') || lower.includes('fiber') || lower.includes('hide') || lower.includes('rock') || lower.includes('sand')) return 'Ressource'
   143→  if (lower.includes('metalbar') || lower.includes('plank') || lower.includes('cloth') || lower.includes('leather') || lower.includes('stoneblock')) return 'Mat\u00e9riau'
   144→  if (lower.includes('potion')) return 'Potion'
   145→  if (lower.includes('pork') || lower.includes('chicken') || lower.includes('beet') || lower.includes('cabbage') || lower.includes('potato') || lower.includes('corn') || lower.includes('mushroom')) return 'Nourriture'
   146→  if (lower.includes('mount')) return 'Monture'
   147→  if (lower.includes('rune') || lower.includes('soul') || lower.includes('relic')) return 'Rune'
   148→  if (lower.includes('book') || lower.includes('journal')) return 'Livre'
   149→  if (lower.includes('avalonian')) return 'Avalonien'
   150→  if (lower.includes('royal')) return 'Royal'
   151→  if (lower.includes('artifact')) return 'Art\u00e9fact'
   152→  if (lower.includes('farm')) return 'Farming'
   153→  return 'Autre'
   154→}

   486→  const [trending, setTrending] = useState<TrendingItem[]>([])
   487→  const [totalItems, setTotalItems] = useState(0)
   488→  const [loading, setLoading] = useState(true)
   489→  const [fetching, setFetching] = useState(false)
   490→  const [connected, setConnected] = useState(false)
   491→  const [updateCount, setUpdateCount] = useState(0)
   492→  const esRef = useRef<EventSource | null>(null)
   493→
   494→  // Update \"time ago\" every 5 seconds
   495→  const [, setTick] = useState(0)
   496→  useEffect(() => {
   497→    const interval = setInterval(() => setTick(t => t + 1), 5000)
   498→    return () => clearInterval(interval)
   499→  }, [])
   500→
   501→  // Connect to SSE stream for live push updates
   502→  useEffect(() => {
   503→    let es: EventSource | null = null
   504→    let reconnectTimeout: ReturnType<typeof setTimeout> | null = null
   505→    let retries = 0
   506→
   507→    function connect() {
   508→      es = new EventSource('/api/stream')
   509→      esRef.current = es
   510→
   511→      es.onopen = () => {
   512→        console.log('[SSE] Connected to live stream')
   513→        setConnected(true)
   514→        retries = 0
   515→      }
   516→
   517→      es.onmessage = (event) => {
   518→        try {
   519→          const payload: MarketPayload = JSON.parse(event.data)
   520→          if (payload.type === 'market-update' && payload.data) {
   521→            setGold(payload.gold)
   522→            setTopSelling(payload.data.topSelling)
   523→            setBlackMarket(payload.data.blackMarket)
   524→            setTrending(payload.data.trending)
   525→            setTotalItems(payload.data.totalItemsTracked)
   526→            setLastUpdate(payload.lastFetchTime)
   527→            setUpdateCount(c => c + 1)
   528→            setLoading(false)
   529→            setFetching(false)
   530→          }
   531→        } catch (err) {
   532→          console.error('[SSE] Parse error:', err)
   533→        }
   534→      }
   535→
   536→      es.onerror = () => {
   537→        console.log('[SSE] Disconnected, reconnecting...')
   538→        setConnected(false)
   539→        es?.close()
   540→        retries++
   541→        const delay = Math.min(2000 * Math.pow(1.5, retries - 1), 15000)
   542→        reconnectTimeout = setTimeout(connect, delay)
   543→      }
   544→    }
   545→
   546→    // Fallback: fetch initial data via /api while SSE connects
   547→    fetch('/api')
   548→      .then(r => r.json())
   549→      .then((payload: MarketPayload) => {
   550→        if (payload.type === 'market-update' && payload.data) {
   551→          setGold(payload.gold)
   552→          setTopSelling(payload.data.topSelling)
   553→          setBlackMarket(payload.data.blackMarket)
   554→          setTrending(payload.data.trending)
   555→          setTotalItems(payload.data.totalItemsTracked)
   556→          setLastUpdate(payload.lastFetchTime)
   557→          setLoading(false)
   558→        }
   559→      })
   560→      .catch(() => setLoading(false))
   561→
   562→    // Start SSE connection
   563→    connect()
   564→
   565→    return () => {
   566→      es?.close()
   567→      if (reconnectTimeout) clearTimeout(reconnectTimeout)
   568→    }
   569→  }, [])
   570→
   571→  const handleManualRefresh = () => {
   572→    setFetching(true)
   573→    // Reconnect SSE to force new data
   574→    esRef.current?.close()
   575→    esRef.current = new EventSource('/api/stream')
   576→  }
   577→
   578→  const avgBlackMarketMargin = blackMarket.length > 0
   579→    ? Math.round(blackMarket.reduce((sum, i) => sum + i.marginPercent, 0) / blackMarket.length)
   580→    : 0
   581→
   582→  return (
   583→    <div className="min-h-screen flex flex-col bg-background">
   584→      {/* Header */}
   585→      <header className="sticky top-0 z-50 border-b border-border/50 bg-background/80 backdrop-blur-xl">
   586→        <div className="max-w-7xl mx-auto px-4 sm:px-6 py-3">
   587→          <div className="flex items-center justify-between">
   588→            <div className="flex items-center gap-3">
   589→              <div className="p-2 rounded-lg bg-primary/20 border border-primary/30">
   590→                <ShoppingBag className="h-5 w-5 text-primary" />
   591→              </div>
   592→              <div>
   593→                <h1 className="text-lg sm:text-xl font-bold tracking-tight">
   594→                  Albion Market <span className="text-primary">Tracker</span>
   595→                </h1>
   596→                <p className="text-xs text-muted-foreground hidden sm:block">
   597→                  Donn\u00e9es en temps r\u00e9el du march\u00e9 &amp; black market
   598→                </p>
   599→              </div>
   600→            </div>
   601→            <div className="flex items-center gap-3">
   602→              <LiveIndicator connected={connected} lastUpdate={lastUpdate} updateCount={updateCount} fetching={fetching} />
   603→              <Button
   604→                variant="outline"
   605→                size="icon"
   606→                onClick={handleManualRefresh}
   607→                disabled={fetching}
   608→                title="Rafra\u00eechir"
   609→              >
   610→                <RefreshCw className={`h-4 w-4 ${fetching ? 'animate-spin' : ''}`} />
   611→              </Button>
   612→            </div>
   613→          </div>
   614→        </div>
   615→      </header>
   616→
   617→      {/* Main Content */}
   618→      <main className="flex-1 max-w-7xl mx-auto w-full px-4 sm:px-6 py-6 space-y-6">
   619→        {/* Gold Ticker */}
   620→        <GoldTicker gold={gold} />
   621→
   622→        {/* Stats Row */}
   623→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   624→          <StatCard
   625→            icon={BarChart3}
   626→            label="Items Suivis"
   627→            value={totalItems}
   628→            subtext="sur tous les march\u00e9s"
   629→            color="bg-primary/20 text-primary"
   630→          />
   631→          <StatCard
   632→            icon={TrendingUp}
   633→            label="Top Ventes"
   634→            value={topSelling.length}
   635→            subtext="items les plus actifs"
   636→            color="bg-emerald-500/20 text-emerald-400"
   637→          />
   638→          <StatCard
   639→            icon={Skull}
   640→            label="Black Market"
   641→            value={blackMarket.length}
   642→            subtext="opportunit\u00e9s d\u00e9tect\u00e9es"
   643→            color="bg-red-500/20 text-red-400"
   644→          />
   645→          <StatCard
   646→            icon={Coins}
   647→            label="Marge Moyenne"
   648→            value={`${avgBlackMarketMargin}%`}
   649→            subtext="marge b\u00e9n\u00e9ficiaire BM"
   650→            color="bg-amber-500/20 text-amber-400"
   651→          />
   652→        </div>
   653→
   654→        {loading ? (
   655→          <LoadingSkeleton />
   656→        ) : (
   657→          <Tabs defaultValue="market" className="space-y-4">
   658→            <TabsList className="grid w-full grid-cols-3">
   659→              <TabsTrigger value="market" className="flex items-center gap-2">
   660→                <TrendingUp className="h-4 w-4" />
   661→                <span className="hidden sm:inline">March\u00e9</span>
   662→              </TabsTrigger>
   663→              <TabsTrigger value="blackmarket" className="flex items-center gap-2">
   664→                <Skull className="h-4 w-4" />
   665→                <span className="hidden sm:inline">Black Market</span>
   666→              </TabsTrigger>
   667→              <TabsTrigger value="trending" className="flex items-center gap-2">
   668→                <Activity className="h-4 w-4" />
   669→                <span className="hidden sm:inline">Tendances</span>
   670→              </TabsTrigger>
   671→            </TabsList>
   672→
   673→            <TabsContent value="market" className="space-y-4">
   674→              <div>
   675→                <h2 className="text-lg font-semibold">Items les Plus Recherch\u00e9s & Vendus</h2>
   676→                <p className="text-sm text-muted-foreground">Class\u00e9s par volume d'activit\u00e9 sur tous les march\u00e9s royaux</p>
   677→              </div>
   678→              <ScrollArea className="max-h-[600px]">
   679→                <TopSellingTable items={topSelling} />
   680→              </ScrollArea>
   681→            </TabsContent>
   682→
   683→            <TabsContent value="blackmarket" className="space-y-4">
   684→              <div>
   685→                <h2 className="text-lg font-semibold flex items-center gap-2">
   686→                  <Skull className="h-5 w-5 text-red-400" />
   687→                  Opportunit\u00e9s Black Market
   688→                </h2>
   689→                <p className="text-sm text-muted-foreground">
   690→                  Acheter sur les march\u00e9s royaux et revendre au Black Market de Caerleon
   691→                </p>
   692→              </div>
   693→              <ScrollArea className="max-h-[600px]">
   694→                <BlackMarketTable items={blackMarket} />
   695→              </ScrollArea>
   696→            </TabsContent>
   697→
   698→            <TabsContent value="trending" className="space-y-4">
   699→              <div>
   700→                <h2 className="text-lg font-semibold">Tendances du March\u00e9</h2>
   701→                <p className="text-sm text-muted-foreground">
   702→                  Items les plus recherch\u00e9s avec \u00e9carts de prix inter-villes
   703→                </p>
   704→              </div>
   705→              <TrendingChart items={trending} />
   706→            </TabsContent>
   707→          </Tabs>
   708→        )}
   709→      </main>
   710→
   711→      {/* Footer */}
   712→      <footer className="border-t border-border/50 bg-background/80 backdrop-blur-xl mt-auto">
   713→        <div className="max-w-7xl mx-auto px-4 sm:px-6 py-3">
   714→          <div className="flex flex-col sm:flex-row items-center justify-between gap-2 text-xs text-muted-foreground">
   715→            <p>Donn\u00e9es fournies par{' '}
   716→              <a href="https://www.albion-online-data.com/" target="_blank" rel="noopener noreferrer" className="text-primary hover:underline">
   717→                Albion Online Data Project
   718→              </a>
   719→            </p>
   720→            <p className="flex items-center gap-1.5">
   721→              <Radio className="h-3 w-3 text-emerald-400" />
   722→              Flux SSE live \u2014 mise \u00e0 jour push toutes les 30s
   723→            </p>
   724→          </div>
   725→        </div>
   726→      </footer>
   727→    </div>
   728→  )
