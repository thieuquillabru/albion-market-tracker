   105→   105→function timeAgo(timestamp: number): string {
   106→   106→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
   107→   107→  if (seconds < 10) return "\u00e0 l'instant"
   108→   108→  if (seconds < 60) return `il y a ${seconds}s`
   109→   109→  const minutes = Math.floor(seconds / 60)
   110→   110→  if (minutes < 60) return `il y a ${minutes}min`
   111→   111→  const hours = Math.floor(minutes / 60)
   112→   112→  return `il y a ${hours}h`
   113→   113→}
