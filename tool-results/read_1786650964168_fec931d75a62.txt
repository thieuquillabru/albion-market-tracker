    21→function formatSilver(price: number | null): string {
    22→  if (price === null || price === undefined) return '-'
    23→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    24→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    25→  return price.toString()
    26→}
    27→
    28→function formatDataAge(dateStr: string | null): { text: string; stale: boolean } {
    29→  if (!dateStr) return { text: '-', stale: true }
    30→  const minutes = Math.floor((Date.now() - new Date(dateStr).getTime()) / 60000)
    31→  if (minutes < 5) return { text: '< 5min', stale: false }
    32→  if (minutes < 60) return { text: `${minutes}min`, stale: false }
    33→  const hours = Math.floor(minutes / 60)
    34→  if (hours < 24) return { text: `${hours}h`, stale: true }
    35→  const days = Math.floor(hours / 24)
    36→  return { text: `${days}j`, stale: true }
    37→}

    43→function timeAgo(timestamp: number): string {
    44→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
    45→  if (seconds < 10) return "à l'instant"
    46→  if (seconds < 60) return `il y a ${seconds}s`
    47→  const minutes = Math.floor(seconds / 60)
    48→  if (minutes < 60) return `il y a ${minutes}min`
    49→  const hours = Math.floor(minutes / 60)
    50→  return `il y a ${hours}h`
    51→}

    72→function getItemCategory(name: string): string {
    73→  const lower = name.toLowerCase()
    74→  if (lower.includes('minerai') || lower.includes('bois') || lower.includes('fibre') || lower.includes('peau') || lower.includes('pierre')) return 'Ressource'
    75→  if (lower.includes('lingot') || lower.includes('planche') || lower.includes('tissu') || lower.includes('cuir') || lower.includes('bloc de pierre')) return 'Matériau'
    76→  if (lower.includes('armure') || lower.includes('coiffe') || lower.includes('chaussures')) return 'Armure'
    77→  if (lower.includes('sac')) return 'Équipement'
    78→  if (lower.includes('cape')) return 'Équipement'
    79→  if (lower.includes('rune') || lower.includes('âme') || lower.includes('relique')) return 'Rune'
    80→  if (lower.includes('potion')) return 'Potion'
    81→  return 'Autre'
    82→}

    84→function getRenderId(dataProjectId: string): string {
    85→  // Most Data Project IDs match render IDs directly.
    86→  // Only a few items need remapping.
    87→  const m = dataProjectId.match(/^(T\d+_|)(.+)$/)
    88→  if (!m) return dataProjectId
    89→  const tier = m[1] || ''
    90→  const base = m[2]
    91→  const MAP: Record<string, string> = {
    92→    CLOTH: 'CLOTHITEM',
    93→    LEATHER: 'LEATHERITEM',
    94→  }
    95→  return tier + (MAP[base] || base)
    96→}
