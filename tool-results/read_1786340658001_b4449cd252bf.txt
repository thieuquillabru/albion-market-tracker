   443→   512→   512→        if (!active || payload.type !== 'market-update' || !payload.data) return
   444→   513→   513→

   455→   524→   524→          pollSeqRef.current++

   457→   526→   526→        }
   458→   527→   527→        setConnected(true)
   459→   528→   528→        setFetching(false)
