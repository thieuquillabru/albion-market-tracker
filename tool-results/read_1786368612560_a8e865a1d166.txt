    25→    25→function formatSilver(price: number | null): string {
    26→    26→  if (price === null || price === undefined) return '-'
    27→    27→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    28→    28→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    29→    29→  return price.toString()
    30→    30→}

    33→    33→  return price.toLocaleString('fr-FR')
    34→    34→}

    36→    36→function timeAgo(timestamp: number): string {
    37→    37→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
    38→    38→  if (seconds < 10) return "à l'instant"
    39→    39→  if (seconds < 60) return `il y a ${seconds}s`
    40→    40→  const minutes = Math.floor(seconds / 60)
    41→    41→  if (minutes < 60) return `il y a ${minutes}min`
    42→    42→  const hours = Math.floor(minutes / 60)
    43→    43→  return `il y a ${hours}h`
    44→    44→}

    65→    65→function getItemCategory(name: string): string {
    66→    66→  const lower = name.toLowerCase()
    67→    67→  if (lower.includes('minerai') || lower.includes('bois') || lower.includes('fibre') || lower.includes('peau') || lower.includes('pierre')) return 'Ressource'
    68→    68→  if (lower.includes('lingot') || lower.includes('planche') || lower.includes('tissu') || lower.includes('cuir') || lower.includes('bloc de pierre')) return 'Matériau'
    69→    69→  if (lower.includes('armure') || lower.includes('coiffe') || lower.includes('chaussures')) return 'Armure'
    70→    70→  if (lower.includes('sac')) return 'Équipement'
    71→    71→  if (lower.includes('cape')) return 'Équipement'
    72→    72→  if (lower.includes('rune') || lower.includes('âme') || lower.includes('relique')) return 'Rune'
    73→    73→  if (lower.includes('potion')) return 'Potion'
    74→    74→  return 'Autre'
    75→    75→}

    77→    77→function getRenderId(dataProjectId: string): string {
    78→    78→  // Most Data Project IDs match render IDs directly.
    79→    79→  // Only a few items need remapping.
    80→    80→  const m = dataProjectId.match(/^(T\d+_|)(.+)$/)
    81→    81→  if (!m) return dataProjectId
    82→    82→  const tier = m[1] || ''
    83→    83→  const base = m[2]
    84→    84→  const MAP: Record<string, string> = {
    85→    85→    CLOTH: 'CLOTHITEM',
    86→    86→    LEATHER: 'LEATHERITEM',
    87→    87→  }
    88→    88→  return tier + (MAP[base] || base)
    89→    89→}
