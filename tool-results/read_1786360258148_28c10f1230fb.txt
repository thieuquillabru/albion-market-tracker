    25→function formatSilver(price: number | null): string {
    26→  if (price === null || price === undefined) return '-'
    27→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    28→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    29→  return price.toString()
    30→}

    32→function formatGoldPrice(price: number): string {
    33→  return price.toLocaleString('fr-FR')
    34→}

    36→function timeAgo(timestamp: number): string {
    37→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
    38→  if (seconds < 10) return "à l'instant"
    39→  if (seconds < 60) return `il y a ${seconds}s`
    40→  const minutes = Math.floor(seconds / 60)
    41→  if (minutes < 60) return `il y a ${minutes}min`
    42→  const hours = Math.floor(minutes / 60)
    43→  return `il y a ${hours}h`
    44→}

    46→function getTierColor(name: string): string {
    47→  if (name.includes('T8')) return 'text-amber-400'
    48→  if (name.includes('T7')) return 'text-purple-400'
    49→  if (name.includes('T6')) return 'text-orange-400'
    50→  if (name.includes('T5')) return 'text-sky-400'
    51→  return 'text-foreground'
    52→}

    65→function getItemCategory(name: string): string {
    66→  const lower = name.toLowerCase()
    67→  if (lower.includes('épée') || lower.includes('claymore') || lower.includes('cimeterre')) return 'Arme Blanche'
    68→  if (lower.includes('dague') || lower.includes('katar')) return 'Arme Blanche'
    69→  if (lower.includes('bâton')) return 'Bâton Magique'
    70→  if (lower.includes('arc') || lower.includes('arbalète')) return 'Arme à Distance'
    71→  if (lower.includes('marteau') || lower.includes('masse') || lower.includes('hache') || lower.includes('hallebarde')) return 'Arme de Mêlée'
    72→  if (lower.includes('coiffe') || lower.includes('sac') || lower.includes('cape')) return 'Armure'
    73→  if (lower.includes('minerai') || lower.includes('bois') || lower.includes('fibre') || lower.includes('peau') || lower.includes('pierre')) return 'Ressource'
    74→  if (lower.includes('lingot') || lower.includes('planche') || lower.includes('tissu') || lower.includes('cuir') || lower.includes('bloc de pierre')) return 'Matériau'
    75→  if (lower.includes('runique')) return 'Matériau Runique'
    76→  if (lower.includes('potion')) return 'Potion'
    77→  if (lower.includes('porc') || lower.includes('poulet') || lower.includes('betterave') || lower.includes('chou') || lower.includes('pomme de terre')) return 'Nourriture'
    78→  if (lower.includes('cheval') || lower.includes('bœuf') || lower.includes('cerf')) return 'Monture'
    79→  if (lower.includes('rune') || lower.includes('âme') || lower.includes('relique')) return 'Rune'
    80→  if (lower.includes('livre') || lower.includes('journal')) return 'Livre'
    81→  if (lower.includes('avalonien')) return 'Avalonien'
    82→  if (lower.includes('royal')) return 'Royal'
    83→  if (lower.includes('artéfact')) return 'Artéfact'
    84→  if (lower.includes('ferme')) return 'Farming'
    85→  return 'Autre'
    86→}
