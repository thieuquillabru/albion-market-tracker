    21→function formatSilver(price: number | null): string {
    22→  if (price === null || price === undefined) return '-'
    23→  if (price >= 1000000) return `${(price / 1000000).toFixed(2)}M`
    24→  if (price >= 1000) return `${(price / 1000).toFixed(1)}K`
    25→  return price.toString()
    26→}
    27→
    28→function formatGoldPrice(price: number): string {
    29→  return price.toLocaleString('fr-FR')
    30→}
    31→
    32→function timeAgo(timestamp: number): string {
    33→  const seconds = Math.floor((Date.now() - timestamp) / 1000)
    34→  if (seconds < 10) return "à l'instant"
    35→  if (seconds < 60) return `il y a ${seconds}s`
    36→  const minutes = Math.floor(seconds / 60)
    37→  if (minutes < 60) return `il y a ${minutes}min`
    38→  const hours = Math.floor(minutes / 60)
    39→  return `il y a ${hours}h`
    40→}

    42→function getTierColor(name: string): string {
    43→  if (name.includes('T8')) return 'text-amber-400'
    44→  if (name.includes('T7')) return 'text-purple-400'
    45→  if (name.includes('T6')) return 'text-orange-400'
    46→  if (name.includes('T5')) return 'text-sky-400'
    47→  return 'text-foreground'
    48→}

    50→function getTierBadgeVariant(name: string): 'default' | 'secondary' | 'outline' | 'destructive' {
    51→  if (name.includes('T8')) return 'default'
    52→  if (name.includes('T7')) return 'secondary'
    53→  return 'outline'
    54→}

    56→function getTierLabel(name: string): string {

    61→function getItemCategory(name: string): string {
    62→  const lower = name.toLowerCase()
    63→  if (lower.includes('minerai') || lower.includes('bois') || lower.includes('fibre') || lower.includes('peau') || lower.includes('pierre')) return 'Ressource'
    64→  if (lower.includes('lingot') || lower.includes('planche') || lower.includes('tissu') || lower.includes('cuir') || lower.includes('bloc de pierre')) return 'Matériau'
    65→  if (lower.includes('armure') || lower.includes('coiffe') || lower.includes('chaussures')) return 'Armure'
    66→  if (lower.includes('sac')) return 'Équipement'
    67→  if (lower.includes('cape')) return 'Équipement'
    68→  if (lower.includes('rune') || lower.includes('âme') || lower.includes('relique')) return 'Rune'
    69→  if (lower.includes('potion')) return 'Potion'
    70→  return 'Autre'
    71→}

    73→function getRenderId(dataProjectId: string): string {
    74→  // Most Data Project IDs match render IDs directly.
    75→  // Only a few items need remapping.
    76→  const m = dataProjectId.match(/^(T\d+_|)(.+)$/)
    77→  if (!m) return dataProjectId
    78→  const tier = m[1] || ''
    79→  const base = m[2]
    80→  const MAP: Record<string, string> = {
    81→    CLOTH: 'CLOTHITEM',
    82→    LEATHER: 'LEATHERITEM',
    83→  }
    84→  return tier + (MAP[base] || base)
    85→}

    87→function ItemIcon({ itemId, size = 32 }: { itemId: string; size?: number }) {
    88→  const [error, setError] = useState(false)
    89→  const renderId = getRenderId(itemId)
    90→  const src = `https://render.albiononline.com/v1/item/${renderId}?quality=1`
    91→  const tier = itemId.match(/T(\d)/)
    92→  const tierNum = tier ? parseInt(tier[1]) : 4
    93→  const colors = ['','oklch(0.7 0.15 250)','oklch(0.65 0.2 200)','oklch(0.7 0.18 40)','oklch(0.7 0.18 40)','oklch(0.75 0.2 80)','oklch(0.7 0.2 300)','oklch(0.65 0.22 300)','oklch(0.8 0.18 80)']
    94→  const color = colors[tierNum] || colors[4]
    95→  if (error) {
    96→    return (
    97→      <div
    98→        className="inline-flex items-center justify-center rounded-md border border-border/30 flex-shrink-0"
    99→        style={{ width: size, height: size, minWidth: size, minHeight: size, backgroundColor: color + '33' }}
   100→        title={renderId}
   101→      >
   102→        <span style={{ fontSize: size * 0.45, color, fontWeight: 700, lineHeight: 1 }}>{tier ? tier[1] : '?'}</span>
   103→      </div>
   104→    )
   105→  }
   106→  return (
   107→    <img
   108→      src={src}
   109→      alt=""
   110→      width={size}
   111→      height={size}
   112→      loading="lazy"
   113→      onError={() => setError(true)}
   114→      className="inline-block rounded-md bg-black/30 border border-border/30 flex-shrink-0"
   115→      style={{ imageRendering: 'auto', minWidth: size, minHeight: size }}
   116→    />
   117→  )
   118→}
